# ============================================================ # PATHOAI - AI BASED PATHOLOGY DETECTION SYSTEM # Google Colab | BloodMNIST | PyTorch | Gradio # Single Cell Complete Project # ============================================================ import os import sys import subprocess import warnings warnings.filterwarnings("ignore") print("Installing project libraries...") subprocess.check_call([ sys.executable, "-m", "pip", "install", "-q", "medmnist", "gradio==5.50.0" ]) import torch import torch.nn as nn import torch.optim as optim import numpy as np import gradio as gr import medmnist import tempfile from datetime import datetime from PIL import Image from medmnist import INFO from torch.utils.data import Dataset, DataLoader # ============================================================ # 1. PROJECT CONFIGURATION # ============================================================ PROJECT_NAME = "PathoAI" IMAGE_SIZE = 64 BATCH_SIZE = 128 EPOCHS = 5 LEARNING_RATE = 0.001 DEVICE = torch.device( "cuda" if torch.cuda.is_available() else "cpu" ) print("Device:", DEVICE) print("Loading BloodMNIST dataset...") # ============================================================ # 2. AUTOMATIC DATASET DOWNLOAD # ============================================================ info = INFO["bloodmnist"] DatasetClass = getattr(medmnist, info["python_class"]) train_data = DatasetClass( split="train", download=True, size=IMAGE_SIZE ) val_data = DatasetClass( split="val", download=True, size=IMAGE_SIZE ) test_data = DatasetClass( split="test", download=True, size=IMAGE_SIZE ) CLASS_NAMES = [ info["label"][str(i)] for i in range(len(info["label"])) ] NUM_CLASSES = len(CLASS_NAMES) print("Dataset downloaded successfully!") print("Training:", len(train_data)) print("Validation:", len(val_data)) print("Testing:", len(test_data)) print("Classes:", CLASS_NAMES) # ============================================================ # 3. DATA PREPROCESSING # ============================================================ class BloodDataset(Dataset): def __init__(self, dataset): self.images = dataset.imgs self.labels = dataset.labels.flatten() def __len__(self): return len(self.images) def __getitem__(self, index): image = self.images[index] image = torch.tensor( image.copy(), dtype=torch.float32 ) image = image.permute(2, 0, 1) / 255.0 label = torch.tensor( int(self.labels[index]), dtype=torch.long ) return image, label train_loader = DataLoader( BloodDataset(train_data), batch_size=BATCH_SIZE, shuffle=True, num_workers=0 ) val_loader = DataLoader( BloodDataset(val_data), batch_size=BATCH_SIZE, shuffle=False, num_workers=0 ) test_loader = DataLoader( BloodDataset(test_data), batch_size=BATCH_SIZE, shuffle=False, num_workers=0 ) # ============================================================ # 4. CNN DEEP LEARNING MODEL # ============================================================ class PathologyAI(nn.Module): def __init__(self): super().__init__() self.network = nn.Sequential( nn.Conv2d(3, 32, kernel_size=3, padding=1), nn.BatchNorm2d(32), nn.ReLU(), nn.MaxPool2d(2), nn.Conv2d(32, 64, kernel_size=3, padding=1), nn.BatchNorm2d(64), nn.ReLU(), nn.MaxPool2d(2), nn.Conv2d(64, 128, kernel_size=3, padding=1), nn.BatchNorm2d(128), nn.ReLU(), nn.MaxPool2d(2), nn.Conv2d(128, 256, kernel_size=3, padding=1), nn.BatchNorm2d(256), nn.ReLU(), nn.AdaptiveAvgPool2d((1, 1)), nn.Flatten(), nn.Dropout(0.3), nn.Linear(256, NUM_CLASSES) ) def forward(self, x): return self.network(x) model = PathologyAI().to(DEVICE) criterion = nn.CrossEntropyLoss() optimizer = optim.Adam( model.parameters(), lr=LEARNING_RATE ) # ============================================================ # 5. MODEL EVALUATION # ============================================================ def evaluate(loader): model.eval() correct = 0 total = 0 loss_sum = 0 with torch.no_grad(): for images, labels in loader: images = images.to(DEVICE) labels = labels.to(DEVICE) outputs = model(images) loss = criterion(outputs, labels) predictions = outputs.argmax(1) correct += ( predictions == labels ).sum().item() total += labels.size(0) loss_sum += loss.item() * labels.size(0) accuracy = 100 * correct / total average_loss = loss_sum / total return accuracy, average_loss # ============================================================ # 6. AI MODEL TRAINING # ============================================================ print("\nStarting PathoAI model training...") training_history = [] for epoch in range(EPOCHS): model.train() correct = 0 total = 0 running_loss = 0 for images, labels in train_loader: images = images.to(DEVICE) labels = labels.to(DEVICE) optimizer.zero_grad() outputs = model(images) loss = criterion(outputs, labels) loss.backward() optimizer.step() running_loss += loss.item() predictions = outputs.argmax(1) correct += ( predictions == labels ).sum().item() total += labels.size(0) train_accuracy = 100 * correct / total val_accuracy, val_loss = evaluate(val_loader) training_history.append([ f"Epoch {epoch + 1}", f"{train_accuracy:.2f}%", f"{val_accuracy:.2f}%", f"{val_loss:.4f}" ]) print( f"Epoch {epoch + 1}/{EPOCHS} | " f"Training Accuracy: {train_accuracy:.2f}% | " f"Validation Accuracy: {val_accuracy:.2f}%" ) # ============================================================ # 7. FINAL TESTING # ============================================================ test_accuracy, test_loss = evaluate(test_loader) print("\nTraining completed!") print("Test Accuracy:", round(test_accuracy, 2), "%") os.makedirs("/content/pathoai", exist_ok=True) torch.save( model.state_dict(), "/content/pathoai/pathoai_model.pth" ) # ============================================================ # 8. AI IMAGE PREDICTION # ============================================================ def predict_image(image, history): if image is None: return ( {}, "Please upload a blood cell image.", history, history, None ) try: model.eval() image = image.convert("RGB") image = image.resize( (IMAGE_SIZE, IMAGE_SIZE) ) array = np.asarray( image, dtype=np.float32 ) / 255.0 tensor = torch.tensor(array) tensor = tensor.permute(2, 0, 1) tensor = tensor.unsqueeze(0).to(DEVICE) with torch.no_grad(): outputs = model(tensor) probabilities = torch.softmax( outputs, dim=1 )[0].cpu().numpy() results = { CLASS_NAMES[i]: float(probabilities[i]) for i in range(NUM_CLASSES) } best_index = int(np.argmax(probabilities)) predicted_class = CLASS_NAMES[best_index] confidence = float( probabilities[best_index] * 100 ) current_time = datetime.now().strftime( "%d-%m-%Y %H:%M:%S" ) # Save history history = history or [] history.insert(0, [ current_time, predicted_class, f"{confidence:.2f}%" ]) history = history[:30] # Create report report = f""" PATHOAI - AI PATHOLOGY REPORT ================================= Project: AI-Based Pathology Detection Date: {current_time} Predicted Blood Cell: {predicted_class} Model Confidence: {confidence:.2f}% Model: CNN Deep Learning Dataset: BloodMNIST Image Resolution: {IMAGE_SIZE} x {IMAGE_SIZE} --------------------------------- DISCLAIMER: This is an educational research result. It is NOT a medical diagnosis. AI predictions can be incorrect. Consult a qualified medical professional. """ report_path = "/content/pathoai/latest_report.txt" with open(report_path, "w") as f: f.write(report) result_text = f""" ## 🧬 AI Analysis Complete **Predicted Blood Cell:** {predicted_class} **Model Confidence:** {confidence:.2f}% **AI Architecture:** CNN Deep Learning **Dataset:** BloodMNIST **Image Size:** {IMAGE_SIZE} × {IMAGE_SIZE} --- ### Medical Safety Notice This educational model cannot establish a medical diagnosis. Do not use its predictions for treatment decisions. """ return ( results, result_text, history, history, report_path ) except Exception as error: return ( {}, f"Prediction Error: {str(error)}", history, history, None ) # ============================================================ # 9. CUSTOM DASHBOARD DESIGN # ============================================================ CSS = """ .gradio-container { max-width: 1450px !important; background: #f4f7fc !important; margin: auto !important; } #header { background: white; padding: 22px; border-radius: 14px; border: 1px solid #e5eaf2; margin-bottom: 20px; } #header h1 { color: #2563eb; font-size: 29px; } #header p { color: #718096; } .stat { background: white; border: 1px solid #e5eaf2; border-radius: 13px; padding: 18px; text-align: center; } .stat h2 { color: #2563eb; font-size: 26px; } .stat p { color: #718096; font-size: 13px; } .panel { background: white; border: 1px solid #e5eaf2; border-radius: 13px; padding: 18px; } #analyze { background: #2563eb !important; color: white !important; font-weight: bold; } footer { text-align: center; } """ # ============================================================ # 10.
